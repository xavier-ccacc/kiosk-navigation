<script>
    // 監聽點擊事件
    document.querySelectorAll('.room').forEach(item => {
        item.addEventListener('click', event => {
            const title = event.target.id;
            showInfo(title, "這是關於 " + title + " 的詳細介紹...");
        });
    });

    function showInfo(title, desc) {
        document.getElementById('room-title').innerText = title;
        document.getElementById('room-desc').innerText = desc;
        document.getElementById('info').style.display = 'block';
    }
</script>
